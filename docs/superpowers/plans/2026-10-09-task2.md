```python
"""Extract the competition zips (public_test_release.zip, sample_dataset.zip) into data/raw/.

Accepts the zips themselves, a folder containing them, or the download bundle zip that wraps them.
"""
from __future__ import annotations

import argparse
import shutil
import zipfile
from pathlib import Path

KNOWN = ("public_test_release.zip", "sample_dataset.zip")


def unpack(src: Path, dest: Path) -> list[Path]:
    src, dest = Path(src), Path(dest)
    dest.mkdir(parents=True, exist_ok=True)
    zips = [src] if src.is_file() else sorted(src.rglob("*.zip"))
    out: list[Path] = []
    for z in zips:
        if z.name in KNOWN:
            with zipfile.ZipFile(z) as zf:
                zf.extractall(dest)
            out.append(dest / z.stem)
            continue
        with zipfile.ZipFile(z) as zf:  # bundle: pull out the known inner zips
            inner = [n for n in zf.namelist() if Path(n).name in KNOWN]
            if not inner:
                continue
            tmp = dest / ".bundle"
            for name in inner:
                zf.extract(name, tmp)
                out += unpack(tmp / name, dest)
            shutil.rmtree(tmp, ignore_errors=True)
    return sorted(set(out))


def main() -> None:
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("src", help="zip file or folder with zips (e.g. data/public-test)")
    ap.add_argument("--dest", default="data/raw")
    a = ap.parse_args()
    for p in unpack(Path(a.src), Path(a.dest)):
        print(p)


if __name__ == "__main__":
    main()
```

```python
def _key(x: str) -> str:
    return re.sub(r"\s+", " ", x).strip().casefold()


_BY_KEY = {_key(lab): lab for lab in LABELS}


def normalize_label(x) -> str | None:
    """Canonical label for `x` (case/whitespace-insensitive, like the leaderboard), else None."""
    return _BY_KEY.get(_key(x)) if isinstance(x, str) else None


def normalize_labels(xs) -> list[str]:
    """Valid labels of `xs`, deduplicated, in canonical order; anything that is not a list gives []."""
    if not isinstance(xs, (list, tuple)):
        return []
    found = {normalize_label(x) for x in xs}
    return [lab for lab in LABELS if lab in found]


def to_vector(labels) -> list[int]:
    s = set(labels)
    return [int(lab in s) for lab in LABELS]


def from_vector(v) -> list[str]:
    return [lab for lab, b in zip(LABELS, v) if b]
```

```python
def make_dataset(root: Path, samples, code: dict | None = None) -> Path:
    """samples: [(sample_id, exam_id, labels or None)]; None = unlabelled release (no `output`)."""
    root.mkdir(parents=True, exist_ok=True)
    (root / "exams.json").write_text(json.dumps(EXAMS, ensure_ascii=False), encoding="utf-8")
    types = {e["exam_id"]: e["exam_type"] for e in EXAMS["exams"]}
    rows = []
    for sid, eid, labels in samples:
        code_file = f"submissions/{eid}/{sid}.cpp"
        (root / code_file).parent.mkdir(parents=True, exist_ok=True)
        (root / code_file).write_text((code or {}).get(sid, CODE + f"// {sid}\n"), encoding="utf-8")
        row = {"sample_id": sid, "input": {"exam_id": eid, "exam_type": types[eid], "language": "cpp11",
                                           "code_file": code_file, "compile_log": "", "test_report": None}}
        if labels is not None:
            row["output"] = {"taxonomy_error": labels}
        rows.append(row)
    (root / "task2_error_taxonomy.json").write_text(
        json.dumps({"task": "task2", "samples": rows}, ensure_ascii=False), encoding="utf-8")
    return root


class CharTokenizer:
    """One token per character: exact, deterministic token budgets in tests."""

    def encode(self, text: str) -> list[int]:
        return [ord(c) for c in text]

    def decode(self, ids: list[int]) -> str:
        return "".join(chr(i) for i in ids)


class FakeEncoder:
    """Deterministic pseudo-random unit vectors keyed by text."""

    encoder_id = "fake-encoder"

    def __init__(self, dim: int = 16):
        self.dim = dim
        self.tokenizer = CharTokenizer()
        self.calls = 0

    def embed(self, texts: list[str]) -> np.ndarray:
        self.calls += 1
        rows = []
        for t in texts:
            seed = int(hashlib.sha256(t.encode("utf-8")).hexdigest()[:8], 16)
            v = np.random.default_rng(seed).standard_normal(self.dim).astype(np.float32)
            rows.append(v / np.linalg.norm(v))
        return np.stack(rows) if rows else np.zeros((0, self.dim), np.float32)

    def peak_memory_gb(self):
        return None


class FakeBackend:
    """Scripted LLM backend: pops one text per requested sample per n."""

    name = "fake"

    def __init__(self, texts):
        self.texts = list(texts)
        self.calls: list[dict] = []

    def generate(self, requests, gen):
        out = []
        for r in requests:
            self.calls.append(r)
            out.append([self.texts.pop(0) for _ in range(int(gen.get("n", 1)))])
        return out

    def info(self):
        return {"backend": "fake", "model": "fake-1", "usage": {}}


def make_fake_heads(path: Path, hidden: int = 16) -> Path:

                "logit_scale": torch.tensor(math.log(20.0)),
                "cfg": {"width": 8, "depth": 3, "hidden_size": hidden, "projection_dim": 4,
                        "activation": "gelu", "layernorm": True, "residual": False}}, path)
    return path
```

```python
"""Load Task 2 samples (labelled sample_dataset or the unlabelled public release). Never reads `feedback`."""
from __future__ import annotations

import json
from dataclasses import dataclass, field
from pathlib import Path

from src.data.loader import load_code, resolve_dataset_root
from src.task2.labels import normalize_labels


@dataclass
class Task2Sample:
    sample_id: str
    exam_id: str
    exam_type: str
    language: str
    code_file: str
    code: str
    compile_log: str | None
    test_report: list | None
    labels: list[str] | None = None

    @property
    def has_gold(self) -> bool:
        return self.labels is not None


@dataclass
class Task2Dataset:
    root: Path
    exams: dict[str, dict]
    samples: list[Task2Sample] = field(default_factory=list)

    def by_id(self) -> dict[str, Task2Sample]:
        return {s.sample_id: s for s in self.samples}


def load_task2(root, task_file: str = "task2_error_taxonomy.json") -> Task2Dataset:
    root = resolve_dataset_root(root)
    exams = {e["exam_id"]: e for e in json.loads((root / "exams.json").read_text(encoding="utf-8"))["exams"]}
    payload = json.loads((root / task_file).read_text(encoding="utf-8"))
    samples = []
    for s in payload["samples"]:
        inp = s["input"]
        if inp["exam_id"] not in exams:
            raise KeyError(f"Unknown exam_id={inp['exam_id']} for {s['sample_id']}")
        out = s.get("output") or {}
        labels = None
        if "taxonomy_error" in out:
            raw = out["taxonomy_error"]
            labels = normalize_labels(raw)
            if len(labels) != len(set(raw)):
                raise ValueError(f"{s['sample_id']}: unknown taxonomy label in {raw}")
        samples.append(Task2Sample(
            sample_id=s["sample_id"], exam_id=inp["exam_id"],
            exam_type=inp.get("exam_type") or exams[inp["exam_id"]]["exam_type"],
            language=inp.get("language", "cpp11"), code_file=inp["code_file"],
            code=load_code(root, inp["code_file"]),
            compile_log=inp.get("compile_log") or None, test_report=inp.get("test_report"),
            labels=labels,
        ))
    return Task2Dataset(root=root, exams=exams, samples=samples)
```

```python
"""Task 2 metrics (macro/micro-F1 over the 10 labels) and the predictions.json format check."""
from __future__ import annotations

import json
from pathlib import Path

from src.task2.labels import LABELS, normalize_label, normalize_labels

NAN = float("nan")


def f1_scores(gold: dict[str, list[str]], pred: dict[str, list[str]]) -> dict:
    """Scores on the ids of `gold`. macro_f1 follows sklearn (label absent on both sides -> F1 0);
    macro_f1_skip_absent averages only labels that occur in gold or predictions."""
    ids = sorted(gold)
    per_label, tp_all, fp_all, fn_all = {}, 0, 0, 0
    for lab in LABELS:
        tp = fp = fn = 0
        for sid in ids:
            g, p = lab in gold[sid], lab in (pred.get(sid) or [])
            tp += g and p
            fp += p and not g
            fn += g and not p
        denom = 2 * tp + fp + fn
        per_label[lab] = {"precision": tp / (tp + fp) if tp + fp else None,
                          "recall": tp / (tp + fn) if tp + fn else None,
                          "f1": 2 * tp / denom if denom else None,
                          "support": tp + fn, "n_pred": tp + fp}
        tp_all, fp_all, fn_all = tp_all + tp, fp_all + fp, fn_all + fn
    f1s = [v["f1"] for v in per_label.values()]
    present = [f for f in f1s if f is not None]
    micro_denom = 2 * tp_all + fp_all + fn_all
    return {
        "n": len(ids),
        "macro_f1": sum(f or 0.0 for f in f1s) / len(LABELS),
        "macro_f1_skip_absent": sum(present) / len(present) if present else NAN,
        "micro_f1": 2 * tp_all / micro_denom if micro_denom else NAN,
        "exact_match": (sum(set(gold[s]) == set(pred.get(s) or []) for s in ids) / len(ids)) if ids else NAN,
        "per_label": per_label,
    }


def validate_predictions(preds, expected_ids=None) -> list[str]:
    """Errors in a Task 2 predictions.json payload (empty list = valid)."""
    if not isinstance(preds, list):
        return ["top-level must be a list"]
    errs, seen = [], set()
    for i, p in enumerate(preds):
        sid = p.get("sample_id") if isinstance(p, dict) else None
        if not sid:
            errs.append(f"[{i}] missing sample_id")
            continue
        if sid in seen:
            errs.append(f"{sid}: duplicate sample_id")
        seen.add(sid)
        labels = (p.get("output") or {}).get("taxonomy_error")
        if not isinstance(labels, list):
            errs.append(f"{sid}: taxonomy_error must be a list")
            continue
        for lab in labels:
            if normalize_label(lab) is None:
                errs.append(f"{sid}: unknown label {lab!r}")
    if expected_ids is not None:
        expected = set(expected_ids)
        if expected - seen:
            errs.append(f"missing {len(expected - seen)} sample_ids, e.g. {sorted(expected - seen)[:5]}")
        if seen - expected:
            errs.append(f"unexpected {len(seen - expected)} sample_ids, e.g. {sorted(seen - expected)[:5]}")
    return errs


def to_submission(labels_by_id: dict[str, list[str]], ids: list[str]) -> list[dict]:
    """One record per id (missing prediction -> []), labels canonicalised."""
    return [{"sample_id": sid, "output": {"taxonomy_error": normalize_labels(labels_by_id.get(sid) or [])}}
            for sid in ids]


def write_submission(path, labels_by_id: dict[str, list[str]], ids: list[str]) -> list[dict]:
    sub = to_submission(labels_by_id, ids)
    errs = validate_predictions(sub, ids)
    if errs:
        raise ValueError(f"invalid Task 2 predictions: {errs[:5]}")
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(sub, ensure_ascii=False, indent=1), encoding="utf-8")
    return sub
```

```python
"""Per-label decision thresholds for Task 2 (macro-F1 rewards getting rare labels right)."""
from __future__ import annotations

import numpy as np

GRID = np.round(np.arange(0.05, 0.951, 0.05), 2)


def best_threshold(p, y, grid=GRID, default: float = 0.5) -> float:
    """Smallest grid value with the best F1 on (p, y); `default` when y has no positive."""
    p, y = np.asarray(p, float), np.asarray(y).astype(bool)
    if not y.any():
        return float(default)
    best_t, best_f = float(default), -1.0
    for t in grid:
        pred = p >= t
        tp = int((pred & y).sum())
        fp = int((pred & ~y).sum())
        fn = int((~pred & y).sum())
        f = 2 * tp / (2 * tp + fp + fn)
        if f > best_f + 1e-12:
            best_t, best_f = float(t), f
    return best_t


def tune_thresholds(P, Y, grid=GRID, default: float = 0.5) -> np.ndarray:
    P, Y = np.asarray(P, float), np.asarray(Y).astype(bool)
    return np.array([best_threshold(P[:, j], Y[:, j], grid, default) for j in range(P.shape[1])])


def apply_thresholds(P, thr, min_rate: float = 0.0) -> np.ndarray:
    """P >= thr per label; then every label gets at least ceil(min_rate * n) positives (highest P first)."""
    P = np.asarray(P, float)
    pred = P >= np.asarray(thr, float)[None, :]
    k = int(np.ceil(min_rate * P.shape[0])) if min_rate > 0 else 0
    if k:
        for j in range(P.shape[1]):
            if pred[:, j].sum() < k:
                pred[np.argsort(-P[:, j], kind="stable")[:k], j] = True
    return pred
```

```python
def submission_text(sample, cfg: dict) -> str:

def demo_text(demos: list, cfg: dict) -> str:
    if not demos:
        return ""
    lim = cfg.get("demo_max_code_chars", 3000)
    parts = ["## VÍ DỤ ĐÃ ĐƯỢC GIẢNG VIÊN GÁN NHÃN (đề khác, chỉ tham khảo cách gán nhãn)"]
    for k, d in enumerate(demos, 1):
        code = d.code if len(d.code) <= lim else d.code[:lim] + "\n// ...(cắt bớt)"
        parts.append(f"### Ví dụ {k} (đề {d.exam_id}, {d.exam_type})\n```cpp\n{code}\n```\n"
                     f"Nhãn của giảng viên: {json.dumps(d.labels, ensure_ascii=False)}\n")
    return "\n".join(parts) + "\n"


def build_messages(exam: dict, sample, demos: list, cfg: dict) -> list[dict]:
    user = "\n".join(x for x in [exam_text(exam), demo_text(demos, cfg), submission_text(sample, cfg),
                                  OUTPUT_INSTRUCTION] if x)
    return [{"role": "system", "content": SYSTEM}, {"role": "user", "content": user}]


def select_demos(pool: list, sample, k: int) -> list:
    """Greedy, deterministic: pick labelled demos (never `sample`) that cover the most not-yet-covered labels,
    preferring the same exam type; one error-free demo ([]) counts as half a new label."""
    if k <= 0:
        return []
    cands = sorted((s for s in pool if s.has_gold and s.sample_id != sample.sample_id),
                   key=lambda s: (s.exam_type != sample.exam_type, s.sample_id))
    chosen, covered = [], set()

    def gain(s) -> float:
        if not s.labels:
            return 0.5 if all(c.labels for c in chosen) else 0.0
        return float(len(set(s.labels) - covered))

    while cands and len(chosen) < k:
        best = max(cands, key=gain)  # max keeps the first of equal gains -> stable order
        chosen.append(best)
        covered |= set(best.labels)
        cands.remove(best)
    return chosen
```

```python
"""Task 2 Hướng 1: prompt -> LLM -> parse labels -> self-consistency vote -> probabilities and labels."""
from __future__ import annotations

from src.task2.labels import LABELS, normalize_labels
from src.task2.prompts import RESPONSE_SCHEMA, build_messages, select_demos
from src.utils.json_extractor import extract_json_objects

RETRY_MSG = ("Câu trả lời trên không phải JSON hợp lệ theo schema yêu cầu. "
             "Hãy trả về DUY NHẤT một object JSON hợp lệ, không thêm chữ nào khác.")


def parse_labels(text: str) -> list[str] | None:
    """Labels from the last JSON object with a list `taxonomy_error` (unknown labels dropped); None if none."""
    for obj in extract_json_objects(text or ""):
        if isinstance(obj.get("taxonomy_error"), list):
            return normalize_labels(obj["taxonomy_error"])
    return None


def vote(label_lists: list[list[str]]) -> dict[str, float]:
    n = len(label_lists)
    return {lab: (sum(lab in ls for ls in label_lists) / n if n else 0.0) for lab in LABELS}


class Task2ApiPipeline:
    def __init__(self, dataset, backend, cfg: dict, demo_pool=()):
        self.ds, self.backend, self.cfg = dataset, backend, cfg
        self.prompt_cfg = cfg.get("prompt", {})
        self.gen_cfg = cfg.get("generation", {})
        self.demo_pool = list(demo_pool)

    def build_request(self, sample) -> dict:
        demos = select_demos(self.demo_pool, sample, int(self.prompt_cfg.get("few_shot_k", 0)))
        exam = self.ds.exams[sample.exam_id]
        return {"sample": sample, "exam": exam, "demos": [d.sample_id for d in demos],
                "messages": build_messages(exam, sample, demos, self.prompt_cfg),
                "schema": RESPONSE_SCHEMA, "schema_name": "task2_labels"}

    def infer(self, samples: list) -> list[dict]:
        reqs = [self.build_request(s) for s in samples]
        outs = self.backend.generate(reqs, self.gen_cfg)
        records = [{"sample_id": q["sample"].sample_id, "exam_id": q["sample"].exam_id, "demos": q["demos"],
                    "outputs": texts, "retry_outputs": []} for q, texts in zip(reqs, outs)]
        for _ in range(int(self.cfg.get("retry_on_parse_fail", 1))):
            bad = [i for i, r in enumerate(records)
                   if all(parse_labels(t) is None for t in r["outputs"] + r["retry_outputs"])]
            if not bad:
                break
            retry = []
            for i in bad:
                q = dict(reqs[i])
                last = (records[i]["retry_outputs"] or records[i]["outputs"] or [""])[-1]
                if last.strip():  # answered but not valid JSON -> ask to fix the format
                    q["messages"] = q["messages"] + [{"role": "assistant", "content": last[-3000:]},
                                                     {"role": "user", "content": RETRY_MSG}]
                retry.append(q)
            gen = {**self.gen_cfg, "n": 1, "temperature": 0.0, "enable_thinking": False}
            for i, texts in zip(bad, self.backend.generate(retry, gen)):
                records[i]["retry_outputs"].extend(texts)
        return records

    @staticmethod
    def empty_prediction(sample) -> dict:
        return {"sample_id": sample.sample_id, "exam_id": sample.exam_id, "parse_ok": False, "n_parsed": 0,
                "probs": {lab: 0.0 for lab in LABELS}, "labels": []}

    def postprocess(self, record: dict) -> dict:
        parsed = [p for p in map(parse_labels, record["outputs"]) if p is not None]
        if not parsed:
            parsed = [p for p in map(parse_labels, record.get("retry_outputs", [])) if p is not None]
        if not parsed:
            return {**self.empty_prediction(self.ds.by_id()[record["sample_id"]])}
        probs = vote(parsed)
        return {"sample_id": record["sample_id"], "exam_id": record["exam_id"], "parse_ok": True,
                "n_parsed": len(parsed), "probs": probs, "labels": [lab for lab in LABELS if probs[lab] >= 0.5]}
```

```python
"""Run Task 2 Hướng 1 on a list of samples: cached/resumable inference, predictions, silver labels, metrics."""
from __future__ import annotations

import hashlib
import json
import time
from pathlib import Path

from src.task1.experiment import REGISTRY, _git_commit, _write_json, append_registry
from src.task2.evaluator import f1_scores, write_submission
from src.task2.pipeline import Task2ApiPipeline
from src.utils.seed import set_seed


def _hash(obj) -> str:
    return hashlib.sha256(json.dumps(obj, sort_keys=True, ensure_ascii=False).encode()).hexdigest()[:16]


def run_api(cfg: dict, dataset, samples: list, backend, out_dir, demo_pool=(), registry: Path | None = REGISTRY,
            use_cache: bool = True) -> dict:
    seed = int(cfg.get("seed", 42))
    set_seed(seed)
    cfg = {**cfg, "generation": {**cfg.get("generation", {}), "seed": seed}}
    out_dir = Path(out_dir)
    out_dir.mkdir(parents=True, exist_ok=True)
    pipe = Task2ApiPipeline(dataset, backend, cfg, demo_pool)
    info = backend.info()
    model_key = {k: info.get(k) for k in ("backend", "model")}
    keys = {s.sample_id: _hash({"model": model_key, "messages": pipe.build_request(s)["messages"],
                                "generation": cfg["generation"], "retry": cfg.get("retry_on_parse_fail", 1)})
            for s in samples}

    cache_path = out_dir / "raw_outputs.jsonl"
    done: dict[str, dict] = {}
    if use_cache and cache_path.exists():
        for line in cache_path.read_text(encoding="utf-8").splitlines():
            row = json.loads(line)
            if keys.get(row["sample_id"]) == row["key"] and pipe.postprocess(row["record"])["parse_ok"]:
                done[row["sample_id"]] = row["record"]
    todo = [s for s in samples if s.sample_id not in done]
    print(f"[{cfg['name']}] {len(done)} bài từ cache, gọi model cho {len(todo)} bài", flush=True)

    usage_before = dict(info.get("usage") or {})
    ledger_before = len(getattr(backend, "ledger", []))
    t0 = time.time()
    chunk = max(1, int(cfg.get("checkpoint_every", 4)))
    try:
        for k in range(0, len(todo), chunk):
            for r in pipe.infer(todo[k:k + chunk]):
                done[r["sample_id"]] = r
                with cache_path.open("a", encoding="utf-8") as f:
                    f.write(json.dumps({"sample_id": r["sample_id"], "key": keys[r["sample_id"]], "record": r},
                                       ensure_ascii=False) + "\n")
    finally:
        if hasattr(backend, "ledger"):
            with (out_dir / "usage.jsonl").open("w", encoding="utf-8") as f:
                for entry in backend.ledger[ledger_before:]:
                    f.write(json.dumps(entry, ensure_ascii=False) + "\n")
    infer_seconds = time.time() - t0

    preds = [pipe.postprocess(done[s.sample_id]) if s.sample_id in done else pipe.empty_prediction(s)
             for s in samples]
    ids = [s.sample_id for s in samples]
    labels = {p["sample_id"]: p["labels"] for p in preds}
    write_submission(out_dir / "predictions.json", labels, ids)
    _write_json(out_dir / "predictions_full.json", preds)
    _write_json(out_dir / "silver_labels.json", {p["sample_id"]: p["probs"] for p in preds if p["parse_ok"]})
    gold = {s.sample_id: s.labels for s in samples if s.has_gold}
    metrics = {"name": cfg["name"], "n": len(samples), "parse_failures": sum(not p["parse_ok"] for p in preds),
               "systems": {"api": f1_scores(gold, labels)} if gold else {}}
    usage_after = backend.info().get("usage") or {}
    delta = {k: usage_after[k] - usage_before.get(k, 0) for k in usage_after
             if isinstance(usage_after[k], (int, float))}
    resources = {"backend": info.get("backend"), "model": info.get("model"), "hardware": "API",
                 "n_samples": len(samples), "seconds_per_sample": infer_seconds / len(todo) if todo else None,
                 "train_seconds": None, "peak_memory_gb": None, "tokens_in": delta.get("prompt_tokens"),
                 "tokens_out": delta.get("output_tokens"), "cost_usd": delta.get("cost_usd")}
    _write_json(out_dir / "metrics.json", metrics)
    _write_json(out_dir / "resources.json", resources)
    _write_json(out_dir / "config.json", {"config": cfg, "model_info": backend.info(),
                                          "timestamp": time.strftime("%Y-%m-%d %H:%M:%S"), "ids": ids})
    if registry is not None:
        api = metrics["systems"].get("api") or {}
        append_registry({"experiment_id": cfg["name"], "timestamp": time.strftime("%Y-%m-%d %H:%M:%S"),
                         "task": "task2", "model": info.get("model", ""), "config_path": cfg.get("_config_path", ""),
                         "metrics": json.dumps({"macro_f1": api.get("macro_f1"), "micro_f1": api.get("micro_f1"),
                                                "n": len(samples), "parse_failures": metrics["parse_failures"]}),
                         "git_commit": _git_commit(), "notes": cfg.get("description", "")}, registry)
    return metrics
```

```python
"""Texts the CLM encoder sees for Task 2, in the layout its heads were trained on (context first, question last).

Each (submission, label) pair is a CLM `noul` question: state = context + "\\n\\n" + question; candidates =
"false: No. This is false: <question>" / "true: Yes. This is true: <question>" (clm.schema.candidates).
The encoder keeps only the first `max_tokens` tokens, so we cut the context ourselves to keep the question.
"""
from __future__ import annotations

from src.task2.labels import DESCRIPTIONS, LABELS

CUT = "\n// ...(cắt bớt)...\n"
SLACK = 8  # tokens of margin for BPE merges across concatenation boundaries


def label_question(label: str) -> str:
    return f'Bài nộp trên có mắc lỗi "{label}" không? {label}: {DESCRIPTIONS[label]}'


def candidate_texts(label: str) -> tuple[str, str]:
    q = label_question(label)
    return f"false: No. This is false: {q}", f"true: Yes. This is true: {q}"


def head_truncate(text: str, n: int, tok) -> str:
    ids = tok.encode(text)
    if len(ids) <= n:
        return text
    return tok.decode(ids[:max(0, n - len(tok.encode(CUT)))]) + CUT


def middle_truncate(text: str, n: int, tok) -> str:
    ids = tok.encode(text)
    if len(ids) <= n:
        return text
    keep = max(0, n - len(tok.encode(CUT)))
    head = keep // 2
    return tok.decode(ids[:head]) + CUT + tok.decode(ids[len(ids) - (keep - head):])


def context_text(exam: dict, sample, tok, budget: int, max_statement_tokens: int = 512) -> str:
    """Exam header + (truncated) statement + student code, at most ~`budget` tokens. No compile_log/test_report:
    the public test has neither, so training and test inputs look alike."""
    head = f"Đề bài {exam['exam_id']} ({exam['exam_type']}, môn {exam.get('course', '')}):\n"
    statement = head_truncate(exam.get("statement", "").strip(), max_statement_tokens, tok)
    mid = "\n\nBài nộp C++ của sinh viên:\n"
    used = len(tok.encode(head + statement + mid))
    return head + statement + mid + middle_truncate(sample.code, max(32, budget - used), tok)


def build_state_texts(exam: dict, sample, tok, max_tokens: int = 2048, max_statement_tokens: int = 512) -> dict:
    questions = {lab: label_question(lab) for lab in LABELS}
    q_tokens = max(len(tok.encode("\n\n" + q)) for q in questions.values())
    ctx_q = context_text(exam, sample, tok, max_tokens - q_tokens - SLACK, max_statement_tokens)
    return {"context": context_text(exam, sample, tok, max_tokens - SLACK, max_statement_tokens),
            "questions": {lab: f"{ctx_q}\n\n{q}" for lab, q in questions.items()}}
```

```python
"""Frozen Qwen3-8B encoder for CLM: last-token hidden state after the final norm, float32, L2-normalised.

Two backends producing the same kind of vectors:
  * MLXEncoder — Apple Silicon, the 4-bit `mlx-community/CLM-v0.1-8B-MLX-4bit` encoder (logic follows that
    repo's clm_mlx/encoder.py, Apache-2.0: head truncation, right padding, last real token);
  * HFEncoder  — CUDA (Kaggle), `Qwen/Qwen3-8B` via transformers.
CachedEncoder stores every embedding on disk, so re-runs and new heads never re-encode.
"""
from __future__ import annotations

import hashlib
import time
from pathlib import Path

import numpy as np


class HFTokenizer:
    def __init__(self, tok):
        self.tok = tok

    def encode(self, text: str) -> list[int]:
        return list(self.tok(text, add_special_tokens=True)["input_ids"])

    def decode(self, ids: list[int]) -> str:
        return self.tok.decode(ids, skip_special_tokens=False)


def _l2(v: np.ndarray) -> np.ndarray:
    return (v / (np.linalg.norm(v, axis=-1, keepdims=True) + 1e-12)).astype(np.float32)


class CachedEncoder:
    def __init__(self, encoder, cache_dir):
        self.encoder = encoder
        self.cache_dir = Path(cache_dir)
        self.cache_dir.mkdir(parents=True, exist_ok=True)
        self.encoder_id = encoder.encoder_id
        self.tokenizer = encoder.tokenizer
        self.n_hits = self.n_computed = 0
        self.seconds = 0.0

    def _path(self, text: str) -> Path:
        key = hashlib.sha256(f"{self.encoder_id}\n{text}".encode("utf-8")).hexdigest()[:32]
        return self.cache_dir / key[:2] / f"{key}.npy"

    def embed(self, texts: list[str]) -> np.ndarray:
        vecs: dict[str, np.ndarray] = {}
        todo = []
        for t in dict.fromkeys(texts):
            p = self._path(t)
            if p.exists():
                vecs[t] = np.load(p)
                self.n_hits += 1
            else:
                todo.append(t)
        if todo:
            t0 = time.time()
            got = _l2(np.asarray(self.encoder.embed(todo), dtype=np.float32))
            self.seconds += time.time() - t0
            self.n_computed += len(todo)
            for t, v in zip(todo, got):
                p = self._path(t)
                p.parent.mkdir(parents=True, exist_ok=True)
                np.save(p, v)
                vecs[t] = v
        if not texts:
            return np.zeros((0, 0), np.float32)
        return np.stack([vecs[t] for t in texts])

    def peak_memory_gb(self):
        fn = getattr(self.encoder, "peak_memory_gb", None)
        return fn() if fn else None


class MLXEncoder:
    def __init__(self, model_dir, max_tokens: int = 2048, batch_tokens: int = 4096):
        import mlx.core as mx
        import mlx.nn as nn
        from mlx_lm import load
        from transformers import AutoTokenizer

        self.mx = mx
        loaded = load(str(model_dir), lazy=True)
        model = loaded[0]
        if hasattr(model, "lm_head"):
            model.lm_head = nn.Identity()  # never used; skip loading it
        model.eval()
        mx.eval(model.model.parameters())
        self.model, self.max_tokens, self.batch_tokens = model, max_tokens, batch_tokens
        self.tokenizer = HFTokenizer(AutoTokenizer.from_pretrained(str(model_dir)))
        self.encoder_id = f"mlx4bit:{Path(model_dir).resolve().parent.name}:{max_tokens}"

    def _ids(self, text: str) -> list[int]:
        ids = self.tokenizer.encode(text) or self.tokenizer.encode(" ")
        return ids[:self.max_tokens]  # head truncation, as the reference vLLM server

    def embed(self, texts: list[str]) -> np.ndarray:
        mx = self.mx
        id_lists = [self._ids(t) for t in texts]
        order = sorted(range(len(id_lists)), key=lambda i: len(id_lists[i]))
        out: list = [None] * len(id_lists)
        i = 0
        while i < len(order):
            j = i + 1
            while j < len(order) and (j - i + 1) * len(id_lists[order[j]]) <= self.batch_tokens:
                j += 1
            rows = [id_lists[k] for k in order[i:j]]
            width = len(rows[-1])
            x = mx.array([r + [0] * (width - len(r)) for r in rows], dtype=mx.int32)  # right padding
            h = self.model.model(x)  # [b, L, hidden] after the final norm
            last = mx.array([len(r) - 1 for r in rows], dtype=mx.int32)
            idx = mx.broadcast_to(last[:, None, None], (len(rows), 1, h.shape[-1]))
            v = mx.take_along_axis(h, idx, axis=1)[:, 0].astype(mx.float32)
            mx.eval(v)
            arr = np.array(v)
            for k, r in zip(order[i:j], range(len(rows))):
                out[k] = arr[r]
            i = j
        return _l2(np.stack(out))

    def peak_memory_gb(self):
        mx = self.mx
        fn = getattr(mx, "get_peak_memory", None) or mx.metal.get_peak_memory
        return fn() / 2**30


class HFEncoder:
    def __init__(self, model_id: str = "Qwen/Qwen3-8B", max_tokens: int = 2048, dtype: str = "float16",
                 load_in_8bit: bool = False, batch_size: int = 2):
        import torch
        from transformers import AutoModel, AutoTokenizer

        self.torch = torch
        kw = {"device_map": "auto"}
        if load_in_8bit:
            from transformers import BitsAndBytesConfig
            kw["quantization_config"] = BitsAndBytesConfig(load_in_8bit=True)
        else:
            kw["torch_dtype"] = getattr(torch, dtype)
        self.model = AutoModel.from_pretrained(model_id, **kw).eval()
        self.tokenizer = HFTokenizer(AutoTokenizer.from_pretrained(model_id))
        self.max_tokens, self.batch_size = max_tokens, batch_size
        self.encoder_id = f"hf:{model_id}:{'int8' if load_in_8bit else dtype}:{max_tokens}"

    def embed(self, texts: list[str]) -> np.ndarray:
        torch = self.torch
        ids = [(self.tokenizer.encode(t) or self.tokenizer.encode(" "))[:self.max_tokens] for t in texts]
        order = sorted(range(len(ids)), key=lambda i: len(ids[i]))
        device = self.model.get_input_embeddings().weight.device
        out: list = [None] * len(ids)
        with torch.no_grad():
            for b in range(0, len(order), self.batch_size):
                idx = order[b:b + self.batch_size]
                rows = [ids[i] for i in idx]
                width = max(map(len, rows))
                x = torch.tensor([r + [0] * (width - len(r)) for r in rows], device=device)
                mask = torch.tensor([[1] * len(r) + [0] * (width - len(r)) for r in rows], device=device)
                h = self.model(input_ids=x, attention_mask=mask).last_hidden_state
                last = torch.tensor([len(r) - 1 for r in rows], device=h.device)
                v = h[torch.arange(len(rows), device=h.device), last].float().cpu().numpy()
                if not np.isfinite(v).all():
                    raise RuntimeError("non-finite embeddings (fp16 overflow?) — set encoder.load_in_8bit: true "
                                       "or dtype: bfloat16 in the config")
                for i, row in zip(idx, v):
                    out[i] = row
        return _l2(np.stack(out))

    def peak_memory_gb(self):
        torch = self.torch
        return sum(torch.cuda.max_memory_allocated(i) for i in range(torch.cuda.device_count())) / 2**30
```

```python
"""CLM projection heads (PyTorch) and the two Task 2 learners on frozen Qwen3-8B embeddings.

(A) NoulModel: CLM's yes/no scoring per label, logit = scale * (cos(S(s), A(c_true)) - cos(S(s), A(c_false))),
    i.e. P(true) of the 2-way softmax CLM uses for `noul` questions. Initialised from a CLM checkpoint and
    fine-tuned with weighted BCE; the encoder stays frozen.
(B) LinearProbe: one logistic regression per label on the context embedding (ablation).
"""
from __future__ import annotations

import copy
import json
from dataclasses import dataclass
from pathlib import Path

import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F


class Head(nn.Module):
    """hidden -> width -> [width -> width]* -> proj MLP; same parameter names as clm.heads.make_head."""

    def __init__(self, hidden: int, width: int, depth: int = 2, proj: int = 512, activation: str = "gelu",
                 layernorm: bool = False, residual: bool = False):
        super().__init__()
        act = {"gelu": nn.GELU, "relu": nn.ReLU, "silu": nn.SiLU}[activation]
        self.inp = nn.Linear(hidden, width)
        self.hidden = nn.ModuleList(nn.Linear(width, width) for _ in range(depth - 2))
        self.norms = nn.ModuleList((nn.LayerNorm(width) if layernorm else nn.Identity()) for _ in range(depth - 2))
        self.out = nn.Linear(width, proj)
        self.act = act()
        self.residual = residual

    def forward(self, x):
        x = self.act(self.inp(x))
        for lin, nrm in zip(self.hidden, self.norms):
            h = self.act(nrm(lin(x)))
            x = x + h if self.residual else h
        return self.out(x)


def _head_kwargs(cfg: dict, top: dict) -> dict:
    return dict(hidden=int(cfg.get("hidden_size", top.get("hidden_size", 4096))), width=int(cfg["width"]),
                depth=int(cfg["depth"]), proj=int(top.get("projection_dim", cfg.get("projection_dim", 512))),
                activation=cfg.get("activation", "gelu"), layernorm=bool(cfg.get("layernorm", False)),
                residual=bool(cfg.get("residual", False)))


def load_heads(path) -> dict:
    """CLM checkpoint: torch `.pt` (upstream release) or a `heads/` dir with config.json + safetensors (MLX port)."""
    p = Path(path)
    if p.is_dir():
        from safetensors.torch import load_file

        top = json.loads((p / "config.json").read_text(encoding="utf-8"))
        cfg = top["cfg"]
        files = sorted(p.glob("*.safetensors"))
        if not files:
            raise FileNotFoundError(f"no .safetensors in {p}")
        w = load_file(str(files[0]))
        sd = {n: {k[len(n) + 1:]: v for k, v in w.items() if k.startswith(n + ".")}
              for n in ("state_head", "action_head")}
        logit = float(w["logit_scale"].float().reshape(-1)[0])
    else:
        ck = torch.load(p, map_location="cpu", weights_only=True)
        top, cfg = ck, dict(ck["cfg"])
        sd = {"state_head": ck["state_head"], "action_head": ck["action_head"]}
        logit = float(torch.as_tensor(ck["logit_scale"]).float().reshape(-1)[0])
    kw = _head_kwargs(cfg, top)
    heads = {}
    for n in ("state_head", "action_head"):
        h = Head(**kw)
        h.load_state_dict({k: v.float() for k, v in sd[n].items()}, strict=True)
        heads[n] = h.float()
    return {"state": heads["state_head"], "action": heads["action_head"], "logit_scale": logit, "cfg": kw}


class NoulModel(nn.Module):
    def __init__(self, state: Head, action: Head, logit_scale: float):
        super().__init__()
        self.state, self.action = state, action
        self.logit_scale = nn.Parameter(torch.tensor(float(logit_scale)))

    @classmethod
    def from_checkpoint(cls, path) -> "NoulModel":
        h = load_heads(path)
        return cls(h["state"], h["action"], h["logit_scale"])

    def scale(self):
        return self.logit_scale.exp().clamp(max=100.0)  # upstream clamps the InfoNCE scale at 100

    def forward(self, s, c_true, c_false):
        zs = F.normalize(self.state(F.normalize(s, dim=-1)), dim=-1)
        zt = F.normalize(self.action(F.normalize(c_true, dim=-1)), dim=-1)
        zf = F.normalize(self.action(F.normalize(c_false, dim=-1)), dim=-1)
        return self.scale() * ((zs * zt).sum(-1) - (zs * zf).sum(-1))


@dataclass
class TrainConfig:
    epochs: int = 15
    lr: float = 1e-5
    weight_decay: float = 0.01
    batch_size: int = 32
    pos_weight_max: float = 10.0
    train_action_head: bool = True
    seed: int = 42
    device: str = "cpu"


def pos_weights(Y, max_w: float) -> torch.Tensor:
    """Inverse-frequency weight of positives per label, in [1, max_w] (soft labels count fractionally)."""
    Y = torch.as_tensor(Y, dtype=torch.float32)
    pos = Y.sum(0)
    return ((Y.shape[0] - pos) / pos.clamp(min=1.0)).clamp(1.0, max_w)


def train_noul(model: NoulModel, S, C, Y, cfg: TrainConfig, eval_fn=None) -> dict:
    torch.manual_seed(cfg.seed)
    dev = torch.device(cfg.device)
    model.to(dev)
    S_t = torch.as_tensor(S, dtype=torch.float32)
    C_t = torch.as_tensor(C, dtype=torch.float32, device=dev)
    Y_t = torch.as_tensor(Y, dtype=torch.float32)
    n, L, _ = S_t.shape
    pw = pos_weights(Y_t, cfg.pos_weight_max).to(dev)
    model.action.requires_grad_(cfg.train_action_head)
    opt = torch.optim.AdamW([p for p in model.parameters() if p.requires_grad], lr=cfg.lr,
                            weight_decay=cfg.weight_decay)
    gen = torch.Generator().manual_seed(cfg.seed)
    hist: dict = {"loss": [], "eval": [], "best_epoch": None}
    best_score, best_state = float("-inf"), None
    for epoch in range(cfg.epochs):
        model.train()
        total = 0.0
        for b in torch.randperm(n * L, generator=gen).split(cfg.batch_size):
            i, j = b // L, b % L
            y = Y_t[i, j].to(dev)
            jd = j.to(dev)
            logit = model(S_t[i, j].to(dev), C_t[jd, 1], C_t[jd, 0])
            loss = F.binary_cross_entropy_with_logits(logit, y, weight=1 + (pw[jd] - 1) * y)
            opt.zero_grad()
            loss.backward()
            opt.step()
            total += loss.item() * len(b)
        hist["loss"].append(total / (n * L))
        if eval_fn is not None:
            score = float(eval_fn(model))
            hist["eval"].append(score)
            if score > best_score:
                best_score, best_state, hist["best_epoch"] = score, copy.deepcopy(model.state_dict()), epoch
    if best_state is not None:
        model.load_state_dict(best_state)
    model.eval()
    return hist


@torch.no_grad()
def predict_noul(model: NoulModel, S, C, batch_size: int = 256) -> np.ndarray:
    model.eval()
    dev = next(model.parameters()).device
    S_t = torch.as_tensor(S, dtype=torch.float32)
    C_t = torch.as_tensor(C, dtype=torch.float32, device=dev)
    n, L, d = S_t.shape
    if n == 0:
        return np.zeros((0, L), np.float32)
    flat, j_all = S_t.reshape(n * L, d), torch.arange(n * L) % L
    out = []
    for k in range(0, n * L, batch_size):
        j = j_all[k:k + batch_size].to(dev)
        out.append(torch.sigmoid(model(flat[k:k + batch_size].to(dev), C_t[j, 1], C_t[j, 0])).cpu())
    return torch.cat(out).reshape(n, L).numpy()


class LinearProbe:
    def __init__(self, C: float = 1.0, seed: int = 42):
        self.C, self.seed = C, seed
        self.models: list = []

    def fit(self, X, Y) -> "LinearProbe":
        from sklearn.linear_model import LogisticRegression

        Yb = np.asarray(Y) >= 0.5
        self.models = []
        for j in range(Yb.shape[1]):
            col = Yb[:, j]
            if col.all() or not col.any():
                self.models.append(float(col.any()))  # constant: no contrast to learn from
            else:
                self.models.append(LogisticRegression(C=self.C, class_weight="balanced", max_iter=2000,
                                                      random_state=self.seed).fit(X, col))
        return self

    def predict_proba(self, X) -> np.ndarray:
        X = np.asarray(X)
        cols = [np.full(len(X), m, np.float32) if isinstance(m, float) else m.predict_proba(X)[:, 1]
                for m in self.models]
        return np.stack(cols, axis=1).astype(np.float32)
```

```python
import json
```

```python
import numpy as np
import torch

SYSTEMS = ("clm_zero_shot", "clm_noul", "probe")


def embed_candidates(encoder) -> np.ndarray:
    texts = [t for lab in LABELS for t in candidate_texts(lab)]
    return encoder.embed(texts).reshape(len(LABELS), 2, -1)


def embed_samples(samples, exams, encoder, text_cfg: dict, dim: int) -> dict:
    if not samples:
        return {"X": np.zeros((0, dim), np.float32), "S": np.zeros((0, len(LABELS), dim), np.float32)}
    contexts, questions = [], []
    for s in samples:
        st = build_state_texts(exams[s.exam_id], s, encoder.tokenizer, **text_cfg)
        contexts.append(st["context"])
        questions.extend(st["questions"][lab] for lab in LABELS)
    return {"X": encoder.embed(contexts), "S": encoder.embed(questions).reshape(len(samples), len(LABELS), -1)}


def _select_public(public, per_exam):
    samples = sorted(public.samples, key=lambda s: s.sample_id)
    if not per_exam:
        return samples
    seen: dict[str, int] = {}
    out = []
    for s in samples:
        if seen.get(s.exam_id, 0) < int(per_exam):
            out.append(s)
            seen[s.exam_id] = seen.get(s.exam_id, 0) + 1
    return out


def _gold_matrix(samples) -> np.ndarray:
    return np.array([to_vector(s.labels) for s in samples], np.float32).reshape(len(samples), len(LABELS))


def run_clm(cfg: dict, encoder, heads_path, gold, public, split: dict, out_dir, silver: dict | None = None,
            registry: Path | None = REGISTRY) -> dict:
    seed = int(cfg.get("seed", 42))
    set_seed(seed)
    out_dir = Path(out_dir)
    out_dir.mkdir(parents=True, exist_ok=True)
    text_cfg = cfg.get("text", {})
    by_id = gold.by_id()
    train_g = [by_id[i] for i in split["train"] if i in by_id]
    val_g = [by_id[i] for i in split["val"] if i in by_id]
    pub = _select_public(public, (cfg.get("data") or {}).get("public_per_exam")) if public else []

    # 1. embeddings (frozen encoder, cached on disk)
    t0 = time.time()
    C = embed_candidates(encoder)
    dim = C.shape[-1]
    e_train = embed_samples(train_g, gold.exams, encoder, text_cfg, dim)
    e_val = embed_samples(val_g, gold.exams, encoder, text_cfg, dim)
    e_pub = embed_samples(pub, public.exams if public else {}, encoder, text_cfg, dim)
    embed_seconds = time.time() - t0

    # 2. training / tuning sets: gold train (+ silver fit) / gold val (+ silver holdout)
    S_tr, X_tr, Y_tr = e_train["S"], e_train["X"], _gold_matrix(train_g)
    S_tu, X_tu, Y_tu = e_val["S"], e_val["X"], _gold_matrix(val_g)
    if silver and pub:
        have = [k for k, s in enumerate(pub) if s.sample_id in silver]
        perm = np.random.default_rng(seed).permutation(have).astype(int)
        n_hold = int(round(len(perm) * float((cfg.get("silver") or {}).get("holdout", 0.1))))
        hold, fit = perm[:n_hold], perm[n_hold:]
        Ys = np.array([[float(silver.get(s.sample_id, {}).get(lab, 0.0)) for lab in LABELS] for s in pub],
                      np.float32)
        S_tr, X_tr, Y_tr = (np.concatenate([S_tr, e_pub["S"][fit]]), np.concatenate([X_tr, e_pub["X"][fit]]),
                            np.concatenate([Y_tr, Ys[fit]]))
        S_tu, X_tu, Y_tu = (np.concatenate([S_tu, e_pub["S"][hold]]), np.concatenate([X_tu, e_pub["X"][hold]]),
                            np.concatenate([Y_tu, Ys[hold]]))
    n_val = len(val_g)
    gold_val = {s.sample_id: s.labels for s in val_g}

    def score(P, thr) -> dict:
        pred = apply_thresholds(P[:n_val], thr)
        return f1_scores(gold_val, {s.sample_id: from_vector(r) for s, r in zip(val_g, pred)})

    # 3. learners
    zero = NoulModel.from_checkpoint(heads_path)
    model = NoulModel.from_checkpoint(heads_path)
    tcfg = TrainConfig(**{**(cfg.get("train") or {}), "seed": seed})
    eval_fn = (lambda m: score(predict_noul(m, e_val["S"], C), np.full(len(LABELS), 0.5))["macro_f1"]) \
        if n_val else None
    t1 = time.time()
    history = train_noul(model, S_tr, C, Y_tr, tcfg, eval_fn)
    probe = LinearProbe(C=float((cfg.get("probe") or {}).get("C", 1.0)), seed=seed).fit(X_tr, Y_tr)
    train_seconds = time.time() - t1

    def probs(system: str, e: dict) -> np.ndarray:
        if system == "probe":
            return probe.predict_proba(e["X"]) if len(e["X"]) else np.zeros((0, len(LABELS)), np.float32)
        return predict_noul(zero if system == "clm_zero_shot" else model, e["S"], C)

    # 4. thresholds + metrics on the gold validation samples
    th = cfg.get("thresholds") or {}
    tune = {"X": X_tu, "S": S_tu}
    thresholds, systems = {}, {}
    for system in SYSTEMS:
        P = probs(system, tune)
        thr = tune_thresholds(P, Y_tu >= 0.5, default=float(th.get("default", 0.5)))
        thresholds[system] = thr.tolist()
        if n_val:
            systems[f"{system}@0.5"] = score(P, np.full(len(LABELS), 0.5))
            systems[f"{system}@tuned"] = score(P, thr)

    # 5. public predictions with the final system
    final = cfg.get("final_system", "clm_noul")
    if pub:
        P_pub = {system: probs(system, e_pub) for system in SYSTEMS}
        pred = apply_thresholds(P_pub[final], np.array(thresholds[final]), float(th.get("min_rate", 0.0)))
        labels = {s.sample_id: from_vector(r) for s, r in zip(pub, pred)}
        write_submission(out_dir / "predictions.json", labels, [s.sample_id for s in pub])
        _write_json(out_dir / "predictions_full.json", [
            {"sample_id": s.sample_id, "exam_id": s.exam_id, "labels": labels[s.sample_id],
             "probs": {system: dict(zip(LABELS, np.round(P_pub[system][k], 4).tolist())) for system in SYSTEMS}}
            for k, s in enumerate(pub)])

    torch.save(model.state_dict(), out_dir / "clm_noul_heads.pt")
    n_samples = len(train_g) + len(val_g) + len(pub)
    peak = encoder.peak_memory_gb() if hasattr(encoder, "peak_memory_gb") else None
    resources = {"backend": "clm", "model": getattr(encoder, "encoder_id", ""), "hardware": hardware_info(),
                 "n_samples": n_samples,
                 "seconds_per_sample": (getattr(encoder, "seconds", embed_seconds) / n_samples) if n_samples else None,
                 "embed_seconds_total": embed_seconds, "n_encoded": getattr(encoder, "n_computed", None),
                 "n_cache_hits": getattr(encoder, "n_hits", None), "train_seconds": train_seconds,
                 "peak_memory_gb": peak, "tokens_in": None, "tokens_out": None, "cost_usd": 0.0}
    metrics = {"name": cfg["name"], "n_train": int(len(S_tr)), "n_val_gold": n_val, "systems": systems,
               "history": history, "final_system": final}
    _write_json(out_dir / "metrics.json", metrics)
    _write_json(out_dir / "thresholds.json", {s: dict(zip(LABELS, t)) for s, t in thresholds.items()})
    _write_json(out_dir / "resources.json", resources)
    _write_json(out_dir / "config.json", {"config": cfg, "heads": str(heads_path),
                                          "timestamp": time.strftime("%Y-%m-%d %H:%M:%S"),
                                          "train_ids": [s.sample_id for s in train_g],
                                          "val_ids": [s.sample_id for s in val_g],
                                          "public_ids": [s.sample_id for s in pub]})
    if registry is not None:
        best = systems.get(f"{final}@tuned") or {}
        append_registry({"experiment_id": cfg["name"], "timestamp": time.strftime("%Y-%m-%d %H:%M:%S"),
                         "task": "task2", "model": resources["model"], "config_path": cfg.get("_config_path", ""),
                         "train_loss": history["loss"][-1] if history["loss"] else "",
                         "metrics": json.dumps({"macro_f1": best.get("macro_f1"), "micro_f1": best.get("micro_f1"),
                                                "n_train": metrics["n_train"], "n_val_gold": n_val}),
                         "git_commit": _git_commit(), "notes": cfg.get("description", "")}, registry)
    return metrics
```

```python
def build_encoder(enc_cfg: dict, max_tokens: int):
    """-> (encoder, heads_path). Downloads only weights/config files, never Python code from the Hub."""
    if enc_cfg["backend"] == "mlx":
        from huggingface_hub import snapshot_download

        from src.task2.clm.encoder import MLXEncoder

        local = Path(snapshot_download(enc_cfg["repo"], local_dir=str(ROOT / enc_cfg["local_dir"]),
                                       allow_patterns=["encoder/*", "heads/*", "clm_mlx.json", "LICENSE", "NOTICE"]))
        return MLXEncoder(local / "encoder", max_tokens, int(enc_cfg.get("batch_tokens", 4096))), local / "heads"
    from huggingface_hub import hf_hub_download

    from src.task2.clm.encoder import HFEncoder

    heads = hf_hub_download(enc_cfg["heads_repo"], enc_cfg["heads_file"])
    enc = HFEncoder(enc_cfg["model"], max_tokens, enc_cfg.get("dtype", "float16"),
                    bool(enc_cfg.get("load_in_8bit", False)), int(enc_cfg.get("batch_size", 2)))
    return enc, Path(heads)

    ap = argparse.ArgumentParser()
    ap.add_argument("--config", default=str(ROOT / "configs" / "task2" / "clm_local.yaml"))
    ap.add_argument("--silver", help="silver_labels.json from run_task2_api.py (overrides config)")
```

```python
COLUMNS = [("run", "Run"), ("system", "Hệ thống"), ("hardware", "Phần cứng"), ("n_eval", "n val"),
           ("macro_f1", "macro-F1"), ("micro_f1", "micro-F1"), ("seconds_per_sample", "giây/bài"),

           ("tokens_out", "token ra"), ("cost_usd", "USD")]


def collect_rows(run_dirs) -> list[dict]:
    rows = []
    for d in map(Path, run_dirs):
        if not (d / "metrics.json").exists() or not (d / "resources.json").exists():
            continue
        metrics = json.loads((d / "metrics.json").read_text(encoding="utf-8"))
        res = json.loads((d / "resources.json").read_text(encoding="utf-8"))
        for system, m in (metrics.get("systems") or {}).items():
            rows.append({"run": d.name, "system": system, "hardware": res.get("hardware"), "n_eval": m.get("n"),
                         "macro_f1": m.get("macro_f1"), "micro_f1": m.get("micro_f1"),
                         **{k: res.get(k) for k in ("seconds_per_sample", "train_seconds", "peak_memory_gb",
                                                    "tokens_in", "tokens_out", "cost_usd")}})
    return rows


def _fmt(key: str, v) -> str:
    if v is None:
        return "—"
    if isinstance(v, float):
        if v != v:  # NaN
            return "—"
        return f"{v:.3f}" if key in ("macro_f1", "micro_f1") else (f"{v:.4f}" if key == "cost_usd" else f"{v:.2f}")
    return str(v)


def to_markdown(rows: list[dict]) -> str:
    head = "| " + " | ".join(t for _, t in COLUMNS) + " |"
    sep = "|" + "|".join("---" for _ in COLUMNS) + "|"
    body = ["| " + " | ".join(_fmt(k, r.get(k)) for k, _ in COLUMNS) + " |" for r in rows]
    return "\n".join([head, sep, *body]) + "\n"
```

## Hướng 2 — CLM-8B
CLM-8B = Qwen3-8B frozen (last-token pooling) + state/action head. Mỗi nhãn là một câu hỏi có/không (`noul`):
P(nhãn) = sigmoid(scale · (cos(state, "true: …") − cos(state, "false: …"))). Hệ thống:
`clm_zero_shot` (head gốc), `clm_noul` (fine-tune head, BCE có trọng số), `probe` (logistic regression trên
embedding context — ablation). Ngưỡng riêng từng nhãn + sàn tỉ lệ dự đoán 2%.

- Local (Mac, MLX 4-bit, tối thiểu): `python scripts/run_task2_clm.py --config configs/task2/clm_local.yaml`
- Kaggle (T4×2, đầy đủ): `kaggle/task2_clm.ipynb` (cần Secret `PUBLIC_TEST_DRIVE_ID`).
- Benchmark: `python scripts/task2_benchmark.py` → `outputs/task2/benchmark.md`.

Head chỉ hợp lệ với encoder đã sinh embedding: bản MLX 4-bit và bản HF fp16 phải train head riêng; số đo hai môi trường
không so trực tiếp.
